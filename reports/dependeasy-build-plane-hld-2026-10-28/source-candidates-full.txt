126      /**
127       * The diff for [paths] alone: what one turn did, rather than how the checkout differs from HEAD.
128       *
129       * [capture] answers the second question, which is the right one for a candidate and useless for
130       * review — in a repository carrying other uncommitted work it buries a twenty-line change under
131       * deleted images and unrelated config. A reviewer who cannot cheaply see what a task changed
132       * ends up trusting the executor's description of it, which is the one thing this seat exists to
133       * avoid.
134       *
135       * Still measured against HEAD, so a file the turn edited that the operator had *also* edited
136       * before it started shows both changes. Only a task-local overlay fixes that; this fixes the
137       * much larger half, which is the hundred files the turn never touched at all.
138       */
139      fun diffOf(paths: List<String>): ScopedDiff {
140          val notices = mutableListOf<String>()
141          val wanted = paths.filter { it.isNotBlank() }.distinct().sorted()
142          if (wanted.isEmpty()) return ScopedDiff("", emptyList())
143          val bounded = wanted.take(SCOPE_LIMIT)
144          if (wanted.size > bounded.size) notices += "Diff scoped to the first ${bounded.size} of ${wanted.size} changed paths"
145          val untracked = (git(*(listOf("ls-files", "-z", "--others", "--exclude-standard", "--") + inputPaths).toTypedArray()) ?: "")
146              .split(NUL).filter { it.isNotEmpty() }.toSet()
147          val base = git("rev-parse", "HEAD")?.trim() ?: "--cached"
148          val text = StringBuilder()
149          // Chunked because a path list is argv, and a turn that touches hundreds of files would
150          // otherwise run into the platform's argument limit and produce nothing at all.
151          bounded.filter { it !in untracked }.chunked(ARGV_CHUNK).forEach { chunk ->
152              val part = git(*(listOf("diff", "--no-ext-diff", "--no-textconv", "--no-color", base, "--") + chunk + CandidateSourceInputs.exclusions(root)).toTypedArray())
153              if (part == null) notices += "Diff unavailable for ${chunk.size} tracked ${if (chunk.size == 1) "path" else "paths"}"
154              else text.append(part)
155          }
156          bounded.filter { it in untracked }.forEach { name ->
157              val addition = git("diff", "--no-index", "--no-ext-diff", "--no-textconv", "--no-color", "--", "/dev/null", name)
158              if (addition == null) notices += "New-file diff unavailable: $name" else text.append(addition)
159          }
160          return ScopedDiff(text.toString(), notices)
161      }

163      /**
164       * A fingerprint of every file git currently considers dirty, for telling one turn's work apart.
165       *
166       * [capture] answers "how does this tree differ from HEAD", which is the right question for a
167       * candidate and the wrong one for a report: in a checkout carrying other uncommitted work it
168       * names a hundred files the agent never touched, and the planner is told the agent changed them
169       * all. Comparing two of these instead answers "what did *this* turn do".
170       *
171       * Only the dirty set is fingerprinted, because a file the agent edits necessarily becomes dirty:
172       * a clean file it touches enters the set, a dirty file it edits changes digest, one it reverts
173       * leaves. Fingerprinting the whole tree would also be correct, and far too slow to run twice a
174       * cycle on a real repository.
175       */
176      fun dirtyManifest(): SourceManifest {
177          val status = git(*(listOf("status", "--porcelain=v1", "-z", "--untracked-files=all", "--") + inputPaths).toTypedArray())
178              ?: return SourceManifest(emptyMap(), available = false)
179          // Porcelain v1 with -z emits "XY path" records separated by NUL; a rename or copy adds a
180          // second record holding the original path, which is not itself a changed file.
181          val records = status.split(NUL).filter { it.length > 3 }
182          val paths = mutableListOf<String>()
183          var index = 0
184          while (index < records.size) {
185              val record = records[index]
186              paths += record.substring(3)
187              if (record[0] == 'R' || record[0] == 'C') index++
188              index++
189          }
190          val distinct = paths.distinct().sorted()
191          val entries = distinct.take(MANIFEST_LIMIT).associateWith { name ->
192              val file = File(root, name)
193              when {
194                  Files.isSymbolicLink(file.toPath()) ->
195                      "link:" + runCatching { Files.readSymbolicLink(file.toPath()).toString() }.getOrDefault("?")
196                  !file.exists() -> "absent"
197                  !file.isFile -> "directory"
198                  // Too big to hash twice a cycle; size and mtime still move when it is edited.
199                  file.length() > MANIFEST_FILE_LIMIT -> "large:" + file.length() + ":" + file.lastModified()
200                  else -> contentDigest(file.readBytes())
201              }
202          }
203          return SourceManifest(entries, available = true, truncated = distinct.size > MANIFEST_LIMIT)
204      }

209      private fun git(vararg args: String): String? = runCatching {
210          val process = ProcessBuilder(listOf("git", "--no-optional-locks", "-C", root.canonicalPath) + args)
211              .redirectError(ProcessBuilder.Redirect.DISCARD).start()
212          val reader = java.util.concurrent.CompletableFuture.supplyAsync { process.inputStream.use { it.readNBytes(16_000_001) } }
213          try {
214              val bytes = reader.get(30, TimeUnit.SECONDS)
215              if (bytes.size > 16_000_000) { process.destroyForcibly(); return@runCatching null }
216              if (!process.waitFor(5, TimeUnit.SECONDS) || process.exitValue() !in if ("--no-index" in args) setOf(0, 1) else setOf(0)) null else bytes.decodeToString()
217          } finally { if (process.isAlive) process.destroyForcibly() }
218      }.getOrNull()
