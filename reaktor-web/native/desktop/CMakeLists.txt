# Configure public profiles before engine construction. Fail closed if the pinned source changes.
function(hook FILE ORIGINAL REPLACEMENT MARKER)
    file(READ "${FILE}" TEXT)
    string(FIND "${TEXT}" "${MARKER}" ALREADY)
    if(ALREADY EQUAL -1)
        string(FIND "${TEXT}" "${ORIGINAL}" FOUND)
        if(FOUND EQUAL -1)
            message(FATAL_ERROR "Pinned construction hook no longer matches: ${MARKER}")
        endif()
        string(REPLACE "${ORIGINAL}" "${REPLACEMENT}" TEXT "${TEXT}")
        file(WRITE "${FILE}" "${TEXT}")
    endif()
endfunction()

    hook("${BACKENDS}/win32_edge.hh" "nullptr, userDataFolder, nullptr, m_com_handler"
