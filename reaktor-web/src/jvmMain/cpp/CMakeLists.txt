cmake_minimum_required(VERSION ${DEPE_CMAKE_MINIMUM})
project(ReaktorWebView LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
    message(FATAL_ERROR "iOS uses the Kotlin/Native WKWebView host")
endif()
find_package(JNI REQUIRED COMPONENTS JVM)
if(NOT DEPE_WEBVIEW_SOURCE)
    message(FATAL_ERROR "A prepared WebView source contract is required")
endif()
add_subdirectory("${DEPE_WEBVIEW_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/deps/webview")
if(APPLE)
    enable_language(OBJCXX)
    set(CMAKE_OBJCXX_STANDARD ${CMAKE_CXX_STANDARD})
    add_library(ReaktorWebView SHARED macos.mm jni.mm)
    target_link_libraries(ReaktorWebView PRIVATE "-framework Cocoa" "-framework WebKit")
    target_compile_options(ReaktorWebView PRIVATE -fno-objc-arc -Wall -Wextra -Werror)
elseif(WIN32)
    add_library(ReaktorWebView SHARED windows.cpp jni.cpp)
    target_link_libraries(ReaktorWebView PRIVATE shlwapi uuid)
    target_compile_definitions(ReaktorWebView PRIVATE UNICODE _UNICODE)
    set_target_properties(ReaktorWebView PROPERTIES PREFIX "")
else()
    add_library(ReaktorWebView SHARED linux.cpp jni.cpp)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(WEB_EXTENSION REQUIRED IMPORTED_TARGET webkit2gtk-web-extension-${REAKTOR_WEBVIEW_WEBKITGTK_API})
    add_library(ReaktorWebExtension SHARED linux_extension.cpp)
    target_link_libraries(ReaktorWebExtension PRIVATE PkgConfig::WEB_EXTENSION)
    add_dependencies(ReaktorWebView ReaktorWebExtension)
endif()
if(MSVC)
    target_compile_options(webview_core_static PRIVATE "/FI${CMAKE_CURRENT_SOURCE_DIR}/reaktor_hooks.h")
    target_compile_options(ReaktorWebView PRIVATE /W4 /EHsc)
else()
    target_compile_options(webview_core_static PRIVATE "-include" "${CMAKE_CURRENT_SOURCE_DIR}/reaktor_hooks.h")
endif()
target_include_directories(ReaktorWebView PRIVATE ${JNI_INCLUDE_DIRS})
target_link_libraries(ReaktorWebView PRIVATE webview::core_static)
set_target_properties(ReaktorWebView PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
add_custom_command(TARGET ReaktorWebView POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy "${DEPE_WEBVIEW_SOURCE}/LICENSE" "$<TARGET_FILE_DIR:ReaktorWebView>/webview-LICENSE")
